190QJsonObject Order::toJson() const
191{

203 QJsonArray itemsArray;

205 itemsArray.append(item->toJson());
206 }
207 json["items"] = itemsArray;
208
209 return json;

212void Order::fromJson(const QJsonObject &json)
213{
214 if (json.contains("id"))
215 setId(json["id"].toString());
216 if (json.contains("customerId"))
217 setCustomerId(json["customerId"].toString());
218 if (json.contains("restaurantId"))
219 setRestaurantId(json["restaurantId"].toString());
220 if (json.contains("status"))
221 setStatus(json["status"].toString());
222 if (json.contains("totalPrice"))
223 setTotalPrice(json["totalPrice"].toDouble());
224 if (json.contains("orderTime"))
225 setOrderTime(QDateTime::fromString(json["orderTime"].toString(), Qt::ISODate));
226 if (json.contains("deliveryTime"))
227 setDeliveryTime(QDateTime::fromString(json["deliveryTime"].toString(), Qt::ISODate));
228
229 if (json.contains("items")) {
230 QList<MenuItem*> items;
231 QJsonArray itemsArray = json["items"].toArray();
232 for (const QJsonValue &value : itemsArray) {
233 if (value.isObject()) {
234 items.append(new MenuItem(value.toObject(), this));
235 }
236 }
237 setItems(items);
238 }
239}
